            event.preventDefault();
            tg.openLink(url);

    })();

  <!-- ADMIN RESET SCRIPT -->

  <script>
    (() => {

      /* CHANGE THIS TO YOUR TELEGRAM NUMERIC ID */

      const ADMIN_TELEGRAM_ID = "YOUR_TELEGRAM_NUMERIC_ID";

      /* APP-SPECIFIC STORAGE PREFIX */

      const APP_PREFIX = "fbmini_";

      const adminButton =
        document.getElementById("miniAdminButton");

      const panel =
        document.getElementById("miniAdminPanel");

      const status =
        document.getElementById("adminStatus");

      const user =
        window.Telegram?.WebApp?.initDataUnsafe?.user;

      /* ADMIN CHECK */

      if (
        !user ||
        ADMIN_TELEGRAM_ID === "YOUR_TELEGRAM_NUMERIC_ID" ||
        String(user.id) !== ADMIN_TELEGRAM_ID
      ) {
        return;
      }

      adminButton.style.display = "block";

      /* PANEL OPEN AND CLOSE */

      adminButton.onclick = () => {
        panel.classList.add("open");
      };

      document.getElementById("closeAdmin").onclick = () => {
        panel.classList.remove("open");
      };

      panel.addEventListener("click", event => {
        if (event.target === panel) {
          panel.classList.remove("open");
        }
      });

      function report(message) {
        status.textContent = message;
      }

      function confirmAction(message) {
        return window.confirm(message);
      }

      /* CLEAR APP STORAGE */

      function clearStorage() {

        if (!confirmAction(
          "Clear this app's saved browser data?"
        )) return;

        let removed = 0;

        for (const storage of [
          localStorage,
          sessionStorage
        ]) {

          const keys = [];

          for (let i = 0; i < storage.length; i++) {
            const key = storage.key(i);

            if (key?.startsWith(APP_PREFIX)) {
              keys.push(key);
            }
          }

          keys.forEach(key => {
            storage.removeItem(key);
            removed++;
          });

        }

        report("✓ Removed " + removed + " saved entries.");
      }

      /* CLEAR APP SESSIONS */

      function clearSessions() {

        if (!confirmAction(
          "Clear this app's saved sessions?"
        )) return;

        const keys = [];

        for (let i = 0; i < sessionStorage.length; i++) {
          const key = sessionStorage.key(i);

          if (key?.startsWith(APP_PREFIX)) {
            keys.push(key);
          }
        }

        keys.forEach(key => {
          sessionStorage.removeItem(key);
        });

        report("✓ Removed " + keys.length +
          " session entries.");
      }

      /* CLEAR APP CACHE */

      async function clearCache() {

        if (!confirmAction(
          "Clear this app's named cache entries?"
        )) return;

        try {

          let removed = 0;

          if ("caches" in window) {

            const names = await caches.keys();

            for (const name of names) {

              if (name.startsWith(APP_PREFIX)) {

                if (await caches.delete(name)) {
                  removed++;
                }

              }

            }

          }

          report("✓ Removed " + removed +
            " matching cache entries.");

        } catch (error) {

          report("Cache error: " + error.message);

        }
      }

      /* FULL LOCAL RESET */

      async function fullReset() {

        if (!confirmAction(
          "FULL LOCAL RESET\n\n" +
          "This clears this app's prefixed browser data, " +
          "matching IndexedDB databases and caches.\n\n" +
          "It does not delete other users' data.\n\n" +
          "Continue?"
        )) return;

        report("Starting local reset...");

        try {

          /* LOCAL AND SESSION STORAGE */

          for (const storage of [
            localStorage,
            sessionStorage
          ]) {

            const keys = [];

            for (let i = 0; i < storage.length; i++) {
              const key = storage.key(i);

              if (key?.startsWith(APP_PREFIX)) {
                keys.push(key);
              }
            }

            keys.forEach(key => storage.removeItem(key));

          }

          report("✓ Browser storage checked.");

          /* INDEXEDDB */

          if (indexedDB.databases) {

            const databases = await indexedDB.databases();

            for (const db of databases) {

              if (!db.name?.startsWith(APP_PREFIX)) continue;

              await new Promise((resolve, reject) => {

                const request =
                  indexedDB.deleteDatabase(db.name);

                request.onsuccess = resolve;

                request.onerror = () =>
                  reject(request.error);

                request.onblocked = () =>
                  reject(new Error("Database deletion blocked"));

              });

            }

          }

          report("✓ Matching databases cleared.");

          /* CACHE STORAGE */

          if ("caches" in window) {

            const names = await caches.keys();

            for (const name of names) {

              if (name.startsWith(APP_PREFIX)) {
                await caches.delete(name);
              }

            }

          }

          report("✓ Matching caches cleared.");

          /* RELOAD */

          report("Local reset completed. Reloading...");

          setTimeout(() => {
            location.reload();
          }, 1500);

        } catch (error) {

          report("Reset incomplete: " + error.message);

        }

      }

      /* CONNECT BUTTONS */

      document.getElementById("clearStorage")
        .onclick = clearStorage;

      document.getElementById("clearSessions")
        .onclick = clearSessions;

      document.getElementById("clearCache")
        .onclick = clearCache;

      document.getElementById("fullReset")
        .onclick = fullReset;

      document.getElementById("topReset")
        .onclick = fullReset;

    })();
  </script>
